cmake_minimum_required(VERSION 3.16)
project(enro_factory_camera LANGUAGES CXX)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_executable(factory_camera camera.cc)
target_compile_features(factory_camera PRIVATE cxx_std_17)
target_link_libraries(factory_camera PRIVATE gz-transport13::core gz-msgs10::gz-msgs10)
set(CMAKE_AUTOMOC ON)
cmake_policy(SET CMP0100 NEW)
find_package(Qt5 REQUIRED COMPONENTS Core Quick QuickControls2)
find_package(gz-gui8 REQUIRED)
find_package(gz-rendering8 REQUIRED)
qt5_add_resources(player_resources player_camera.qrc)
add_library(PlayerCamera SHARED PlayerCamera.cc PlayerCamera.hh ${player_resources})
target_compile_features(PlayerCamera PRIVATE cxx_std_17)
target_link_libraries(PlayerCamera PRIVATE gz-gui8::gz-gui8 gz-rendering8::gz-rendering8
  gz-transport13::core gz-msgs10::gz-msgs10 Qt5::Quick Qt5::QuickControls2)
